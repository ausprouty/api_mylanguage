UPDATE i18n_translation_queue
SET status='queued', lockedAt=NULL, lockedBy=NULL, attempts=0, runAfter=UTC_TIMESTAMP()
WHERE status='processing'
  AND (lockedAt IS NULL
       OR lockedAt <= UTC_TIMESTAMP() - INTERVAL 10 MINUTE
       OR lockedAt >  UTC_TIMESTAMP() + INTERVAL 5 MINUTE);

UPDATE i18n_translation_queue
SET status='queued', attempts=0, runAfter=UTC_TIMESTAMP()
WHERE status='failed';

SELECT
  SUM(status='queued'
      AND runAfter<=UTC_TIMESTAMP()
      AND (lockedAt IS NULL
        OR lockedAt <= UTC_TIMESTAMP() - INTERVAL 10 MINUTE
        OR lockedAt >  UTC_TIMESTAMP() + INTERVAL 5 MINUTE)) AS eligible,
  SUM(status='queued')     AS queued,
  SUM(status='processing') AS processing,
  SUM(status='failed')     AS failed
FROM i18n_translation_queue;
